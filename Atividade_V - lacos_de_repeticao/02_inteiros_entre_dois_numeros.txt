#include <stdio.h>
int main(){
    int num1, num2;
    printf("digite o primeiro numero inteiro ");
    scanf("%d",&num1);

    printf("digite o segundo numero inteiro ");
    scanf("%d",&num2);

    if (num1>num2)
    {
        int temp = num1;
        num1 = num2;
        num2 = temp;
    }
    

    for (int i = num1 + 1; i < num2; i++)
    {
        printf("%d\n", i);
    }
    
    return 0;
}
